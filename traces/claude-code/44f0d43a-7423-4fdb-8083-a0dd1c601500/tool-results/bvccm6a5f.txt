     1	/**
     2	 * The pipeline: one deep module behind `buildPipeline(deps)`, the composition
     3	 * root that receives every port plus the injected `clock`, `delay` and id
     4	 * source, and news-up nothing itself (spec, Pipeline). Tests drive it with
     5	 * fakes and an instant `delay`; production passes real timers.
     6	 *
     7	 * Ticket 05 is the skeleton: it walks the Run through `discovering →
     8	 * evaluating → applying → completed` over whatever the `JobSource` returns,
     9	 * marking every Posting `skipped` because there is no evaluator yet. Later
    10	 * tickets replace the evaluating and applying stages with real work.
    11	 */
    12	import { emptyFunnel, jobKey, type Posting, type Repo, type Run } from "@auto-apply/shared";
    13	import { logger } from "../logger.js";
    14	import type { JobSource } from "./ports.js";
    15	
    16	export interface PipelineDeps {
    17	  repo: Repo;
    18	  jobSource: JobSource;
    19	  clock: () => Date;
    20	  /** Paces the stages so the live view is watchable; instant in tests. */
    21	  delay: (ms: number) => Promise<void>;
    22	  newRunId: () => string;
    23	}
    24	
    25	export interface StartedRun {
    26	  runId: string;
    27	  /**
    28	   * Settles when the background work ends, never rejects (a failure becomes a
    29	   * `failed` Run). The route ignores it; tests await it.
    30	   */
    31	  finished: Promise<void>;
    32	}
    33	
    34	export interface Pipeline {
    35	  /**
    36	   * Creates the Run and starts it in the background, resolving as soon as the
    37	   * Run exists. Rejects with `ActiveRunExistsError` if the user already has an
    38	   * active Run (D17).
    39	   */
    40	  startRun(uid: string): Promise<StartedRun>;
    41	}
    42	
    43	/** Pause between skeleton stages, so a demo can watch the statuses move. */
    44	const STEP_MS = 500;
    45	
    46	const SKELETON_REASON = "skeleton pipeline: no evaluator yet";
    47	
    48	export function buildPipeline(deps: PipelineDeps): Pipeline {
    49	  const { repo, jobSource, clock, delay, newRunId } = deps;
    50	
    51	  async function process(runId: string): Promise<void> {
    52	    await delay(STEP_MS);
    53	    const postings = await jobSource.discover();
    54	    for (const posting of postings) await queue(runId, posting);
    55	    await repo.patchRun(runId, { funnelIncrements: { discovered: postings.length } });
    56	
    57	    await repo.patchRun(runId, { status: "evaluating" });
    58	    for (const posting of postings) {
    59	      const key = jobKey(posting);
    60	      await repo.patchEvaluation(runId, key, { status: "evaluating" });
    61	      await delay(STEP_MS);
    62	      await repo.patchEvaluation(runId, key, { status: "skipped", reason: SKELETON_REASON });
    63	      await repo.patchRun(runId, { funnelIncrements: { evaluated: 1, skipped: 1 } });
    64	    }
    65	
    66	    await repo.patchRun(runId, { status: "applying" });
    67	    await delay(STEP_MS);
    68	    await repo.patchRun(runId, { status: "completed" });
    69	  }
    70	
    71	  async function queue(runId: string, posting: Posting): Promise<void> {
    72	    const now = clock().toISOString();
    73	    await repo.putEvaluation(runId, {
    74	      jobKey: jobKey(posting),
    75	      runId,
    76	      posting,
    77	      status: "queued",
    78	      verdict: null,
    79	      score: null,
    80	      reason: null,
    81	      createdAt: now,
    82	      updatedAt: now,
    83	    });
    84	  }
    85	
    86	  /** A crash inside the Run fails the Run with its reason; it never escapes. */
    87	  async function fail(runId: string, err: unknown): Promise<void> {
    88	    const reason = err instanceof Error ? err.message : String(err);
    89	    logger.error("run_failed", { runId, reason });
    90	    try {
    91	      await repo.patchRun(runId, { status: "failed", reason });
    92	    } catch (patchErr) {
    93	      logger.error("run_fail_not_recorded", {
    94	        runId,
    95	        error: patchErr instanceof Error ? patchErr.message : String(patchErr),
    96	      });
    97	    }
    98	  }
    99	
   100	  return {
   101	    async startRun(uid) {
   102	      const now = clock().toISOString();
   103	      const run: Run = {
   104	        runId: newRunId(),
   105	        uid,
   106	        status: "discovering",
   107	        funnel: emptyFunnel(),
   108	        reason: null,
   109	        createdAt: now,
   110	        updatedAt: now,
   111	      };
   112	      await repo.createRun(run);
   113	      const finished = process(run.runId).catch((err: unknown) => fail(run.runId, err));
   114	      return { runId: run.runId, finished };
   115	    },
   116	  };
   117	}

=== apps/api/src/routes/run-stream.ts
     1	/**
     2	 * The live Run stream (D21): one SSE response per connected browser, fed by the
     3	 * `Repo` subscriptions. Every connect waits for the first `watchRun` callback
     4	 * and the first `watchEvaluations` batch, sends them as one `snapshot`, then
     5	 * forwards later states as `run` / `eval` deltas (full documents, ADR-0003).
     6	 * A terminal Run is followed by `done` and the end of the response. There is no
     7	 * `Last-Event-ID`: a reconnect simply re-snapshots.
     8	 */
     9	import type { Request, Response } from "express";
    10	import {
    11	  doneEventSchema,
    12	  errorEventSchema,
    13	  evalEventSchema,
    14	  isRunActive,
    15	  runEventSchema,
    16	  snapshotEventSchema,
    17	  type Evaluation,
    18	  type EvaluationChange,
    19	  type Repo,
    20	  type Run,
    21	  type SseEventName,
    22	} from "@auto-apply/shared";
    23	import { logger } from "../logger.js";
    24	
    25	export function streamRun(
    26	  req: Request,
    27	  res: Response,
    28	  repo: Repo,
    29	  runId: string,
    30	  heartbeatMs: number,
    31	): void {
    32	  res.status(200).set({
    33	    "Content-Type": "text/event-stream",
    34	    "Cache-Control": "no-cache",
    35	    // nginx buffers proxied responses by default; this turns it off per
    36	    // response (CODING_STANDARDS landmine: SSE buffering).
    37	    "X-Accel-Buffering": "no",
    38	    Connection: "keep-alive",
    39	  });
    40	  res.flushHeaders();
    41	
    42	  let closed = false;
    43	  // Until the snapshot is sent, callbacks only accumulate the latest state.
    44	  let run: Run | undefined;
    45	  let evaluations: Map<string, Evaluation> | undefined;
    46	  let snapshotSent = false;
    47	
    48	  const send = (event: SseEventName, data: unknown) => {
    49	    if (!closed) res.write(`event: ${event}\ndata: ${JSON.stringify(data)}\n\n`);
    50	  };
    51	
    52	  const finish = () => {
    53	    if (closed) return;
    54	    closed = true;
    55	    unsubscribeRun();
    56	    unsubscribeEvaluations();
    57	    clearInterval(heartbeat);
    58	    res.end();
    59	  };
    60	
    61	  /** A terminal Run ends the stream: the terminal `run` has gone out, so `done`. */
    62	  const endIfTerminal = (latest: Run) => {
    63	    if (isRunActive(latest.status)) return;
    64	    send("done", doneEventSchema.parse({ runId: latest.runId, status: latest.status }));
    65	    finish();
    66	  };
    67	
    68	  const trySnapshot = () => {
    69	    if (snapshotSent || !run || !evaluations) return;
    70	    snapshotSent = true;
    71	    send("snapshot", snapshotEventSchema.parse({ run, evaluations: [...evaluations.values()] }));
    72	    endIfTerminal(run);
    73	  };
    74	
    75	  const onRun = (latest: Run) => {
    76	    run = latest;
    77	    if (!snapshotSent) return trySnapshot();
    78	    send("run", runEventSchema.parse(latest));
    79	    endIfTerminal(latest);
    80	  };
    81	
    82	  const onEvaluations = (changes: EvaluationChange[]) => {
    83	    if (!snapshotSent) {
    84	      evaluations ??= new Map();
    85	      for (const { type, evaluation } of changes) {
    86	        if (type === "removed") evaluations.delete(evaluation.jobKey);
    87	        else evaluations.set(evaluation.jobKey, evaluation);
    88	      }
    89	      return trySnapshot();
    90	    }
    91	    for (const { type, evaluation } of changes) {
    92	      // Evaluations are never deleted while a Run streams; nothing to show.
    93	      if (type !== "removed") send("eval", evalEventSchema.parse(evaluation));
    94	    }
    95	  };
    96	
    97	  // Never silent (ADR-0003): a subscription failure reaches the browser.
    98	  const onError = (error: Error) => {
    99	    logger.error("run_stream_subscription_error", { runId, error: error.message });
   100	    send(
   101	      "error",
   102	      errorEventSchema.parse({ error: { code: "subscription_error", message: error.message } }),
   103	    );
   104	  };
   105	
   106	  const heartbeat = setInterval(() => {
   107	    if (!closed) res.write(": heartbeat\n\n");
   108	  }, heartbeatMs);
   109	
   110	  const unsubscribeRun = repo.watchRun(runId, onRun, onError);
   111	  const unsubscribeEvaluations = repo.watchEvaluations(runId, onEvaluations, onError);
   112	
   113	  // The browser went away: release the listeners and the timer (landmine:
   114	  // listener leaks).
   115	  req.on("close", finish);
   116	}

    19	/** Run ids are UUIDs; anything else never names a Run (and never a Firestore path). */

    26	export function runsRouter(repo: Repo, pipeline: Pipeline, heartbeatMs: number): Router {
    27	  const router = Router();

    29	  router.post("/api/runs", async (_req, res, next) => {
    30	    try {
    31	      const { runId } = await pipeline.startRun(DEMO_UID);
    32	      res.status(202).json(createRunResponseSchema.parse({ runId }));

    34	      if (err instanceof ActiveRunExistsError) {

    37	      }

    39	    }
    40	  });
    41

    43	    try {
    44	      const run = await repo.getActiveRun(DEMO_UID);

    46	    } catch (err) {
    47	      next(err);
    48	    }
    49	  });
    50	
    51	  router.get("/api/runs/:runId/events", async (req, res, next) => {
    52	    const params = runParamsSchema.safeParse(req.params);
    53	    if (!params.success) {
    54	      sendError(res, 400, "invalid_request", "Malformed run id");
    55	      return;
    56	    }
    57	    try {
    58	      const run = await repo.getRun(params.data.runId);

    61	        return;
    62	      }
    63	      streamRun(req, res, repo, run.runId, heartbeatMs);
    64	    } catch (err) {
    65	      next(err);
    66	    }
    67	  });
    68	
    69	  return router;
    70	}

    61	  override watchRun(runId: string, cb: (run: Run) => void, onError: SubscriptionErrorHandler) {

   113	/** Reads an SSE response frame by frame, as an `EventSource` would. */
   114	async function openStream(app: ReturnType<typeof harness>["app"], runId: string) {
   115	  const server: Server = await new Promise((resolve) => {
   116	    const s = app.listen(0, () => resolve(s));
   117	  });
   118	  servers.push(server);
   119	  const { port } = server.address() as AddressInfo;
   120	  const controller = new AbortController();
   121	  const res = await fetch(`http://127.0.0.1:${port}/api/runs/${runId}/events`, {
   122	    signal: controller.signal,
   123	  });
   124	  const reader = res.body?.pipeThrough(new TextDecoderStream()).getReader();
   125	  let buffer = "";
   126	
   127	  /** The next frame, or `null` once the server has ended the stream. */
   128	  async function next(): Promise<Frame | null> {
   129	    for (;;) {
   130	      const end = buffer.indexOf("\n\n");
   131	      if (end !== -1) {
   132	        const block = buffer.slice(0, end);
   133	        buffer = buffer.slice(end + 2);
   134	        return parseFrame(block);
   135	      }
   136	      const chunk = await reader?.read();
   137	      if (!chunk || chunk.done) return null;
   138	      buffer += chunk.value;
   139	    }
   140	  }
   141	
   142	  async function rest(): Promise<Frame[]> {
   143	    const frames: Frame[] = [];
   144	    for (let frame = await next(); frame; frame = await next()) frames.push(frame);
   145	    return frames;
   146	  }
   147	
   148	  return { res, next, rest, close: () => controller.abort() };
   149	}

   151	function parseFrame(block: string): Frame {
   152	  if (block.startsWith(":")) return { comment: block.slice(1).trim() };
   153	  let event = "message";
   154	  let data = "";
   155	  for (const line of block.split("\n")) {
   156	    if (line.startsWith("event: ")) event = line.slice(7);
   157	    if (line.startsWith("data: ")) data += line.slice(6);
   158	  }
   159	  return { event, data: JSON.parse(data) as unknown };
   160	}

     1	/**
     2	 * The one place the browser holds a Run's live stream (CODING_STANDARDS, Web).
     3	 * It opens an `EventSource` with credentials, parses every event with the
     4	 * contract's schemas, and closes on `done`, on unmount, or when the run id
     5	 * changes. State is always the latest the server sent (ADR-0003): a `snapshot`
     6	 * replaces everything, so a reconnect re-renders from scratch and loses nothing.
     7	 */
     8	import { useEffect, useState } from "react";
     9	import {
    10	  doneEventSchema,
    11	  errorEventSchema,
    12	  evalEventSchema,
    13	  runEventSchema,
    14	  snapshotEventSchema,
    15	  type Evaluation,
    16	  type Run,
    17	} from "@auto-apply/shared";
    18	import { runEventsUrl } from "../api";
    19	
    20	export type Connection = "connecting" | "live" | "reconnecting" | "closed";
    21	
    22	export interface RunStreamState {
    23	  connection: Connection;
    24	  run: Run | null;
    25	  /** The latest Evaluation per Job Key, in the order they first appeared. */
    26	  evaluations: Evaluation[];
    27	  error: string | null;
    28	}
    29	
    30	const initial: RunStreamState = {
    31	  connection: "connecting",
    32	  run: null,
    33	  evaluations: [],
    34	  error: null,
    35	};
    36	
    37	function upsert(evaluations: Evaluation[], next: Evaluation): Evaluation[] {
    38	  const i = evaluations.findIndex((e) => e.jobKey === next.jobKey);
    39	  if (i === -1) return [...evaluations, next];
    40	  return evaluations.map((e, j) => (j === i ? next : e));
    41	}
    42	
    43	export function useRunStream(runId: string | null): RunStreamState {
    44	  const [state, setState] = useState<RunStreamState>(initial);
    45	
    46	  useEffect(() => {
    47	    setState(initial);
    48	    if (!runId) return;
    49	
    50	    const source = new EventSource(runEventsUrl(runId), { withCredentials: true });
    51	
    52	    /** Parses one event's JSON; a malformed one is reported, never trusted. */
    53	    const handle =
    54	      <T>(parse: (data: unknown) => T, apply: (prev: RunStreamState, data: T) => RunStreamState) =>
    55	      (event: Event) => {
    56	        let data: T;
    57	        try {
    58	          data = parse(JSON.parse((event as MessageEvent<string>).data));
    59	        } catch {
    60	          setState((prev) => ({ ...prev, error: `Malformed ${event.type} event from the server` }));
    61	          return;
    62	        }
    63	        setState((prev) => apply(prev, data));
    64	      };
    65	
    66	    source.addEventListener(
    67	      "snapshot",
    68	      handle(snapshotEventSchema.parse, (_prev, { run, evaluations }) => ({
    69	        connection: "live",
    70	        run,
    71	        evaluations,
    72	        error: null,
    73	      })),
    74	    );
    75	    source.addEventListener(
    76	      "run",
    77	      handle(runEventSchema.parse, (prev, run) => ({ ...prev, run })),
    78	    );
    79	    source.addEventListener(
    80	      "eval",
    81	      handle(evalEventSchema.parse, (prev, evaluation) => ({
    82	        ...prev,
    83	        evaluations: upsert(prev.evaluations, evaluation),
    84	      })),
    85	    );
    86	    const onDone = handle(doneEventSchema.parse, (prev) => ({
    87	      ...prev,
    88	      connection: "closed" as const,
    89	    }));
    90	    source.addEventListener("done", (event) => {
    91	      // The server ends the response next; without close() the browser would
    92	      // reconnect and stream the finished Run again.
    93	      source.close();
    94	      onDone(event);
    95	    });
    96	    // `error` is both the server's named event (it carries data) and the
    97	    // browser's own connection-error signal (it does not).
    98	    source.addEventListener("error", (event) => {
    99	      if (event instanceof MessageEvent) {
   100	        handle(errorEventSchema.parse, (prev, { error }) => ({ ...prev, error: error.message }))(
   101	          event,
   102	        );
   103	        return;
   104	      }
   105	      setState((prev) => ({
   106	        ...prev,
   107	        connection: source.readyState === EventSource.CLOSED ? "closed" : "reconnecting",
   108	      }));
   109	    });
   110	
   111	    return () => source.close();
   112	  }, [runId]);
   113	
   114	  return state;
   115	}

     1	/**
     2	 * A stand-in for the browser's `EventSource` (spec, UI seam): tests push
     3	 * contract-shaped server events into it and inspect how it was opened and
     4	 * whether it was closed.
     5	 */
     6	export class FakeEventSource extends EventTarget {
     7	  static readonly CONNECTING = 0;
     8	  static readonly OPEN = 1;
     9	  static readonly CLOSED = 2;
    10	  static instances: FakeEventSource[] = [];
    11	
    12	  readyState = FakeEventSource.CONNECTING;
    13	  readonly withCredentials: boolean;
    14	
    15	  constructor(
    16	    readonly url: string,
    17	    init?: EventSourceInit,
    18	  ) {
    19	    super();
    20	    this.withCredentials = init?.withCredentials ?? false;
    21	    FakeEventSource.instances.push(this);
    22	  }
    23	
    24	  /** The most recently opened stream. */
    25	  static latest(): FakeEventSource {
    26	    const latest = FakeEventSource.instances.at(-1);
    27	    if (!latest) throw new Error("no EventSource was opened");
    28	    return latest;
    29	  }
    30	
    31	  /** The server sends a named event with a JSON payload. */
    32	  emit(event: string, data: unknown): void {
    33	    this.readyState = FakeEventSource.OPEN;
    34	    this.dispatchEvent(new MessageEvent(event, { data: JSON.stringify(data) }));
    35	  }
    36	
    37	  /** The connection drops; the browser will retry (`reconnecting`) unless closed. */
    38	  drop(): void {
    39	    this.readyState = FakeEventSource.CONNECTING;
    40	    this.dispatchEvent(new Event("error"));
    41	  }
    42	
    43	  close(): void {
    44	    this.readyState = FakeEventSource.CLOSED;
    45	  }
